/**
 * usePerformanceProfiler.ts
 * Standalone Mobile Render Profiler for React Native
 * Measures first-click latency & eliminates bottleneck re-renders.
 */
import { useEffect, useRef } from 'react';

export function usePerformanceProfiler(componentName: string) {
  const renderCount = useRef(0);
  const startTime = useRef(performance.now());

  renderCount.current += 1;

  useEffect(() => {
    const duration = performance.now() - startTime.current;
    
    // Emit telemetry if render threshold exceeded (>16.6ms frame budget)
    if (duration > 16.6) {
      console.warn(`[PerfAlert] ⚠️ ${componentName} dropped a frame: ${duration.toFixed(2)}ms (Render #${renderCount.current})`);
    }
  });

  return { renderCount: renderCount.current };
}

/**
 * HermesMemoryMonitor.ts
 * Real-time heap inspection & garbage collection management
 */
export class HermesMemoryMonitor {
  private static thresholdMB = 80;

  static checkHeapUsage(): void {
    if (global.HermesInternal?.getInstrumentedStats) {
      const stats = global.HermesInternal.getInstrumentedStats();
      const allocatedMB = stats.js_allocated_bytes / (1024 * 1024);
      
      if (allocatedMB > this.thresholdMB) {
        // Trigger non-blocking generational GC
        global.gc?.();
      }
    }
  }
}
